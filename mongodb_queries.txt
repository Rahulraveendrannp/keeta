// Set date range variables (adjust dates as needed)
var startDate = ISODate("2025-12-01T00:00:00.000Z");
var endDate = ISODate("2025-12-08T00:00:00.000Z"); // End of Dec 7

// ============================================
// 1. OVERVIEW SUMMARY (Dec 1-7, 2025)
// ============================================

// 1.1 Total Registered Players
db.users.countDocuments({
  createdAt: { $gte: startDate, $lt: endDate }
});

// 1.2 Total Games Played
// Count users who completed games 1-3 (have gameTiers set) + game 4 (have gameClaims.game4 = true)
db.users.aggregate([
  {
    $match: {
      createdAt: { $gte: startDate, $lt: endDate }
    }
  },
  {
    $project: {
      game1Played: { $cond: [{ $ne: ["$gameTiers.game1", null] }, 1, 0] },
      game2Played: { $cond: [{ $ne: ["$gameTiers.game2", null] }, 1, 0] },
      game3Played: { $cond: [{ $ne: ["$gameTiers.game3", null] }, 1, 0] },
      game4Played: { $cond: ["$gameClaims.game4", 1, 0] }
    }
  },
  {
    $project: {
      totalGames: { $add: ["$game1Played", "$game2Played", "$game3Played", "$game4Played"] }
    }
  },
  {
    $group: {
      _id: null,
      totalGamesPlayed: { $sum: "$totalGames" }
    }
  }
]);

// 1.3 Total Prizes Redeemed
db.users.aggregate([
  {
    $match: {
      createdAt: { $gte: startDate, $lt: endDate }
    }
  },
  {
    $project: {

      game2Claimed: { $cond: ["$gameClaims.game2", 1, 0] },

      totalPrizes: { $add: ["$game1Claimed", "$game2Claimed", "$game3Claimed", "$game4Claimed"] }
    }
  },
  {
    $group: {
      _id: null,
      totalPrizesRedeemed: { $sum: "$totalPrizes" }
    }
  }
]);

// 1.4 Tier 1 Prizes Redeemed
db.users.aggregate([
  {
    $match: {
      createdAt: { $gte: startDate, $lt: endDate },

    $project: {
      game1Tier1: { $cond: [{ $and: [{ $eq: ["$gameTiers.game1", 1] }, "$gameClaims.game1"] }, 1, 0] },
      game2Tier1: { $cond: [{ $and: [{ $eq: ["$gameTiers.game2", 1] }, "$gameClaims.game2"] }, 1, 0] },
      game3Tier1: { $cond: [{ $and: [{ $eq: ["$gameTiers.game3", 1] }, "$gameClaims.game3"] }, 1, 0] }

  },
  {
    $project: {
      totalTier1: { $add: ["$game1Tier1", "$game2Tier1", "$game3Tier1"] }
    }
  },
  {
    $group: {
      _id: null,
      tier1PrizesRedeemed: { $sum: "$totalTier1" }
    }
  }
]);

// 2. DAILY PERFORMANCE BREAKDOWN
// ============================================

// For each day (Day 1 = Dec 1, Day 2 = Dec 2, etc.)
// Day 1: Dec 1, 2025
var day1Start = ISODate("2025-12-01T00:00:00.000Z");
var day1End = ISODate("2025-12-02T00:00:00.000Z");

db.users.aggregate([
  {
    $match: {
      createdAt: { $gte: day1Start, $lt: day1End }
    }
  },
  {
    $project: {
      date: { $dateToString: { format: "%Y-%m-%d", date: "$createdAt" } },
      totalUsers: 1,
      game1Played: { $cond: [{ $ne: ["$gameTiers.game1", null] }, 1, 0] },
      game2Played: { $cond: [{ $ne: ["$gameTiers.game2", null] }, 1, 0] },
      game3Played: { $cond: [{ $ne: ["$gameTiers.game3", null] }, 1, 0] },
      game4Played: { $cond: ["$gameClaims.game4", 1, 0] },
      game1Claimed: { $cond: ["$gameClaims.game1", 1, 0] },
      game2Claimed: { $cond: ["$gameClaims.game2", 1, 0] },
      game3Claimed: { $cond: ["$gameClaims.game3", 1, 0] },
      game4Claimed: { $cond: ["$gameClaims.game4", 1, 0] },
      game1Tier1: { $cond: [{ $and: [{ $eq: ["$gameTiers.game1", 1] }, "$gameClaims.game1"] }, 1, 0] },
      game2Tier1: { $cond: [{ $and: [{ $eq: ["$gameTiers.game2", 1] }, "$gameClaims.game2"] }, 1, 0] },
      game3Tier1: { $cond: [{ $and: [{ $eq: ["$gameTiers.game3", 1] }, "$gameClaims.game3"] }, 1, 0] },
      game1Tier2: { $cond: [{ $and: [{ $eq: ["$gameTiers.game1", 2] }, "$gameClaims.game1"] }, 1, 0] },
      game2Tier2: { $cond: [{ $and: [{ $eq: ["$gameTiers.game2", 2] }, "$gameClaims.game2"] }, 1, 0] },
      game3Tier2: { $cond: [{ $and: [{ $eq: ["$gameTiers.game3", 2] }, "$gameClaims.game3"] }, 1, 0] },
      lastQRScanAt: 1
    }
  },
  {
    $group: {
      _id: "$date",
      totalUsers: { $sum: 1 },
      totalGamesPlayed: { $sum: { $add: ["$game1Played", "$game2Played", "$game3Played", "$game4Played"] } },
      totalPrizesRedeemed: { $sum: { $add: ["$game1Claimed", "$game2Claimed", "$game3Claimed", "$game4Claimed"] } },
      tier1PrizeRedeemed: { $sum: { $add: ["$game1Tier1", "$game2Tier1", "$game3Tier1"] } },
      tier2PrizeRedeemed: { $sum: { $add: ["$game1Tier2", "$game2Tier2", "$game3Tier2"] } },
      aiPhotoPrints: { $sum: "$game4Claimed" },
      peakHours: { $push: { $hour: "$lastQRScanAt" } }
    }
  }
]);

// 4. HOURLY TRAFFIC ANALYSIS
// ============================================

// Hourly user counts grouped by time blocks
db.users.aggregate([
  {
    $match: {
      createdAt: { $gte: startDate, $lt: endDate }
    }
  },
  {
    $project: {
      hour: { $hour: "$createdAt" },
      createdAt: 1
    }
  },
  {
    $bucket: {
      groupBy: "$hour",
      boundaries: [0, 14, 15, 18, 21, 24],
      default: "other",
      output: {
        count: { $sum: 1 },
        users: { $push: { hour: "$hour", createdAt: "$createdAt" } }
      }
    }
  },
  {
    $project: {
      timeBlock: {
        $switch: {
          branches: [
            { case: { $eq: ["$_id", 0] }, then: "12AM-2PM (Non-ope)" },
            { case: { $eq: ["$_id", 14] }, then: "2PM-3PM" },
            { case: { $eq: ["$_id", 15] }, then: "3PM-6PM" },
            { case: { $eq: ["$_id", 18] }, then: "6PM-9PM" },
            { case: { $eq: ["$_id", 21] }, then: "9PM-12AM" }
          ],
          default: "other"
        }
      },
      userCount: "$count"
    }
  }
]);

// Alternative: More detailed hourly breakdown
db.users.aggregate([
  {
    $match: {
      createdAt: { $gte: startDate, $lt: endDate }
    }
  },
  {
    $project: {
      hour: { $hour: "$createdAt" },
      timeBlock: {
        $switch: {
          branches: [
            { case: { $lt: ["$hour", 14] }, then: "12AM-2PM (Non-ope)" },
            { case: { $and: [{ $gte: ["$hour", 14] }, { $lt: ["$hour", 15] }] }, then: "2PM-3PM" },
            { case: { $and: [{ $gte: ["$hour", 15] }, { $lt: ["$hour", 18] }] }, then: "3PM-6PM" },
            { case: { $and: [{ $gte: ["$hour", 18] }, { $lt: ["$hour", 21] }] }, then: "6PM-9PM" },
            { case: { $gte: ["$hour", 21] }, then: "9PM-12AM" }
          ],
          default: "other"
        }
      }
    }
  },
  {
    $group: {
      _id: "$timeBlock",
      userCount: { $sum: 1 }
    }
  },
  {
    $sort: {
      _id: 1
    }
  }
]);
